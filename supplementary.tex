\section*{مقدمه}

در این مواد تکمیلی، تمام جزئیات فنی، ریاضی و پیاده‌سازی موتور پردازش اقلیم‌شناسی (ClimateProcessingEngine) به‌صورت کامل و گام‌به‌گام ارائه شده است. هدف این است که خواننده، حتی بدون پیش‌زمینهٔ آماری یا برنامه‌نویسی، بتواند از این سامانه استفاده کند و روش‌های پیاده‌سازی‌شده را درک نماید.

این مواد تکمیلی شامل بخش‌های زیر است:

\begin{enumerate}
\item معرفی کامل مخزن GitHub و نحوهٔ نصب و اجرا
\item تشریح دقیق تابع چگالی احتمال، روش برآورد پارامترها و معیارهای انتخاب برای هر یک از ۵ توزیع
\item توضیح گام‌به‌گام الگوریتم EM برای توزیع دوکوهانه (نوآوری پژوهش)
\item معرفی شاخص‌های تشخیص دوکوهانه‌ای
\item راهنمای عملی برای افزودن توزیع جدید
\item مثال‌های عددی کامل با کد پایتون
\end{enumerate}

\section{تشریح کامل توزیع‌های آماری}

در این بخش، هر یک از پنج توزیع آماری به‌صورت کامل معرفی می‌شوند.

\subsection{توزیع نرمال (Normal Distribution)}

\subsubsection{تعریف}

توزیع نرمال با تابع چگالی احتمال زیر تعریف می‌شود:

\begin{equation}
f(x) = \frac{1}{\sigma\sqrt{2\pi}} \exp\left(-\frac{(x-\mu)^2}{2\sigma^2}\right)
\end{equation}

\subsubsection{پارامترها}

\begin{itemize}
\item $\mu$: میانگین (مکان) – محدوده: $(-\infty, \infty)$
\item $\sigma > 0$: انحراف معیار (مقیاس)
\end{itemize}

\subsubsection{روش برآورد پارامترها}

برآوردگرهای بیشینه‌ی درست‌نمایی:

\begin{equation}
\hat{\mu} = \frac{1}{n}\sum_{i=1}^{n} x_i
\end{equation}

\begin{equation}
\hat{\sigma} = \sqrt{\frac{1}{n}\sum_{i=1}^{n} (x_i - \hat{\mu})^2}
\end{equation}

\subsubsection{تابع درست‌نمایی}

\begin{equation}
\mathcal{L}(\mu, \sigma) = \prod_{i=1}^{n} \frac{1}{\sigma\sqrt{2\pi}} \exp\left(-\frac{(x_i-\mu)^2}{2\sigma^2}\right)
\end{equation}

\subsubsection{لگاریتم درست‌نمایی}

\begin{equation}
\ell(\mu, \sigma) = -\frac{n}{2}\ln(2\pi) - n\ln(\sigma) - \frac{1}{2\sigma^2}\sum_{i=1}^{n} (x_i - \mu)^2
\end{equation}

\subsubsection{کد نمونه در پایتون}

\begin{verbatim}
import numpy as np
from scipy.stats import norm

def fit_normal(data):
    mu = np.mean(data)
    sigma = np.std(data, ddof=0)
    loglik = np.sum(norm.logpdf(data, mu, sigma))
    n = len(data)
    k = 2
    aic = 2*k - 2*loglik
    bic = k*np.log(n) - 2*loglik
    aicc = aic + (2*k*(k+1))/(n-k-1) if n-k-1 > 0 else aic
    return {'mu': mu, 'sigma': sigma, 'loglik': loglik, 
            'aicc': aicc, 'bic': bic}
\end{verbatim}

\subsection{توزیع نرمال چوله (Skew-Normal Distribution)}

\subsubsection{تعریف}

توزیع نرمال چوله با تابع چگالی احتمال زیر تعریف می‌شود:

\begin{equation}
f(x) = \frac{2}{\omega} \phi\left(\frac{x-\xi}{\omega}\right) \Phi\left(\alpha\frac{x-\xi}{\omega}\right)
\end{equation}

که در آن $\phi$ تابع چگالی نرمال استاندارد و $\Phi$ تابع توزیع تجمعی نرمال استاندارد است.

\subsubsection{پارامترها}

\begin{itemize}
\item $\xi$: پارامتر مکان – محدوده: $(-\infty, \infty)$
\item $\omega > 0$: پارامتر مقیاس
\item $\alpha$: پارامتر شکل (چولگی) – محدوده: $(-\infty, \infty)$
\end{itemize}

\subsubsection{روش برآورد پارامترها (روش گشتاورها)}

\begin{enumerate}
\item چولگی نمونه ($\hat{\gamma}_1$) را محاسبه کنید.
\item $\delta$ را از رابطهٔ زیر محاسبه کنید:
\begin{equation}
\delta = \sqrt{\frac{\pi}{2}} \frac{\hat{\gamma}_1}{1 + \frac{2}{\pi}\hat{\gamma}_1^2}
\end{equation}
\item $\alpha$ را محاسبه کنید:
\begin{equation}
\alpha = \frac{\delta}{\sqrt{1-\delta^2}}
\end{equation}
\item $\omega$ را محاسبه کنید:
\begin{equation}
\omega = \frac{\sigma}{\sqrt{1 - \frac{2\delta^2}{\pi}}}
\end{equation}
\item $\xi$ را محاسبه کنید:
\begin{equation}
\xi = \mu - \omega\delta\sqrt{\frac{2}{\pi}}
\end{equation}
\end{enumerate}

\subsubsection{کد نمونه در پایتون}

\begin{verbatim}
import numpy as np
from scipy.stats import skewnorm

def fit_skewnormal(data):
    mu = np.mean(data)
    sigma = np.std(data, ddof=0)
    skew = np.mean(((data - mu)/sigma)**3)
    
    # برآورد اولیه با روش گشتاورها
    delta = np.sqrt(np.pi/2) * skew / (1 + (2/np.pi)*skew**2)
    delta = np.clip(delta, -0.999, 0.999)
    alpha = delta / np.sqrt(1 - delta**2)
    omega = sigma / np.sqrt(1 - 2*delta**2/np.pi)
    xi = mu - omega * delta * np.sqrt(2/np.pi)
    
    # بهینه‌سازی عددی (اختیاری)
    # ...
    
    loglik = np.sum(skewnorm.logpdf(data, alpha, xi, omega))
    n = len(data)
    k = 3
    aic = 2*k - 2*loglik
    bic = k*np.log(n) - 2*loglik
    aicc = aic + (2*k*(k+1))/(n-k-1) if n-k-1 > 0 else aic
    return {'alpha': alpha, 'xi': xi, 'omega': omega, 
            'loglik': loglik, 'aicc': aicc, 'bic': bic}
\end{verbatim}

\subsection{توزیع مقادیر حدی تعمیم‌یافته (GEV)}

\subsubsection{تعریف}

توزیع GEV با تابع چگالی احتمال زیر تعریف می‌شود:

\begin{equation}
f(x) = \frac{1}{\sigma} \left(1 + \xi\frac{x-\mu}{\sigma}\right)^{-\left(\frac{1}{\xi}+1\right)} \exp\left(-\left(1 + \xi\frac{x-\mu}{\sigma}\right)^{-\frac{1}{\xi}}\right)
\end{equation}

برای $\xi \neq 0$، و برای $\xi = 0$:

\begin{equation}
f(x) = \frac{1}{\sigma} \exp\left(-\frac{x-\mu}{\sigma} - \exp\left(-\frac{x-\mu}{\sigma}\right)\right)
\end{equation}

\subsubsection{پارامترها}

\begin{itemize}
\item $\mu$: پارامتر مکان – محدوده: $(-\infty, \infty)$
\item $\sigma > 0$: پارامتر مقیاس
\item $\xi$: پارامتر شکل – محدوده: $(-\infty, \infty)$
\end{itemize}

\subsubsection{محدودهٔ تعریف}

\begin{itemize}
\item $\xi > 0$: $x \in [\mu - \sigma/\xi, \infty)$ (توزیع فرشه)
\item $\xi < 0$: $x \in (-\infty, \mu - \sigma/\xi]$ (توزیع ویبول)
\item $\xi = 0$: $x \in (-\infty, \infty)$ (توزیع گامبل)
\end{itemize}

\subsubsection{روش برآورد پارامترها (روش گشتاورها)}

\begin{enumerate}
\item میانگین ($\mu_x$)، انحراف معیار ($\sigma_x$) و چولگی ($\gamma_1$) داده را محاسبه کنید.
\item $\xi$ را با حل عددی معادلهٔ زیر تخمین بزنید:
\begin{equation}
\gamma_1 = \frac{\Gamma(1-3\xi) - 3\Gamma(1-\xi)\Gamma(1-2\xi) + 2\Gamma(1-\xi)^3}{(\Gamma(1-2\xi) - \Gamma(1-\xi)^2)^{3/2}}
\end{equation}
\item $\sigma$ را محاسبه کنید:
\begin{equation}
\sigma = \frac{|\xi|\sigma_x}{\sqrt{\Gamma(1-2\xi) - \Gamma(1-\xi)^2}}
\end{equation}
\item $\mu$ را محاسبه کنید:
\begin{equation}
\mu = \mu_x + \frac{\sigma(\Gamma(1-\xi)-1)}{\xi}
\end{equation}
\end{enumerate}

\subsubsection{کد نمونه در پایتون}

\begin{verbatim}
import numpy as np
from scipy.stats import genextreme
from scipy.special import gamma

def fit_gev(data):
    mu_x = np.mean(data)
    sigma_x = np.std(data, ddof=0)
    skew = np.mean(((data - mu_x)/sigma_x)**3)
    
    # برآورد اولیه ξ با روش گشتاورها
    # (در عمل، از بهینه‌سازی عددی استفاده می‌شود)
    # ...
    
    # برآورد با استفاده از scipy
    params = genextreme.fit(data)
    c, loc, scale = params[0], params[1], params[2]
    
    loglik = np.sum(genextreme.logpdf(data, c, loc, scale))
    n = len(data)
    k = 3
    aic = 2*k - 2*loglik
    bic = k*np.log(n) - 2*loglik
    aicc = aic + (2*k*(k+1))/(n-k-1) if n-k-1 > 0 else aic
    return {'shape': c, 'loc': loc, 'scale': scale, 
            'loglik': loglik, 'aicc': aicc, 'bic': bic}
\end{verbatim}

\subsection{توزیع دوکوهانه (Bimodal Normal Distribution) – نوآوری پژوهش}

\subsubsection{تعریف}

توزیع دوکوهانه، ترکیبی وزنی از دو توزیع نرمال است:

\begin{equation}
f(x) = w_1 \mathcal{N}(x; \mu_1, \sigma_1^2) + w_2 \mathcal{N}(x; \mu_2, \sigma_2^2)
\end{equation}

که در آن $w_1 + w_2 = 1$ و $0 \leq w_i \leq 1$.

\subsubsection{پارامترها}

\begin{itemize}
\item $w_1, w_2$: وزن‌های مؤلفه‌ها ($w_1 + w_2 = 1$)
\item $\mu_1, \mu_2$: میانگین‌های دو مؤلفه
\item $\sigma_1, \sigma_2 > 0$: انحراف معیارهای دو مؤلفه
\end{itemize}

تعداد کل پارامترها: ۵ (با در نظر گرفتن $w_2 = 1 - w_1$)

\subsubsection{الگوریتم بیشینه‌سازی امید ریاضی (EM)}

الگوریتم EM برای برآورد پارامترهای توزیع دوکوهانه به‌صورت زیر است:

\paragraph{مرحلهٔ ۱: مقداردهی اولیه}

\begin{enumerate}
\item میانگین ($\mu$) و انحراف معیار ($\sigma$) کل داده را محاسبه کنید.
\item چولگی ($\gamma_1$) و کشیدگی ($\gamma_2$) داده را محاسبه کنید.
\item اگر $|\gamma_1| < 0.1$:
  \begin{itemize}
  \item $w_1 = 0.5$، $w_2 = 0.5$
  \item اگر $\gamma_2 > 0$: $\delta = \sigma \sqrt{\max(0, \gamma_2/2)}$ در غیر این صورت $\delta = 0.5\sigma$
  \item $\mu_1 = \mu - \delta$، $\mu_2 = \mu + \delta$
  \item $\sigma_1 = \sigma_2 = \sqrt{\max(0.1, \sigma^2 - \delta^2)}$
  \end{itemize}
\item در غیر این صورت:
  \begin{itemize}
  \item $w_1 = 0.5 - 0.4 \times \text{clip}(\gamma_1, -1, 1)$
  \item $w_2 = 1 - w_1$
  \item $\delta = \sigma \times (0.5 + 0.3 \times |\gamma_1|)$
  \item اگر $\gamma_1 > 0$: $\mu_1 = \mu - w_2\delta$، $\mu_2 = \mu + w_1\delta$
  \item اگر $\gamma_1 < 0$: $\mu_1 = \mu + w_2\delta$، $\mu_2 = \mu - w_1\delta$
  \item $\sigma_1 = \sigma_2 = \sqrt{\max(0.1, \sigma^2 - w_1 w_2 \delta^2)}$
  \end{itemize}
\item $w_1 = \text{clip}(w_1, 0.01, 0.99)$، $w_2 = 1 - w_1$
\item $\sigma_1 = \max(\sigma_1, 0.1)$، $\sigma_2 = \max(\sigma_2, 0.1)$
\end{enumerate}

\paragraph{مرحلهٔ ۲: تکرار EM}

برای هر تکرار تا رسیدن به همگرایی (حداکثر ۱۰۰۰ تکرار یا تغییر پارامترها کمتر از $10^{-6}$):

\begin{enumerate}
\item \textbf{مرحلهٔ E (Expectation):} برای هر مشاهده $x_i$، احتمال تعلق به مؤلفهٔ ۱ را محاسبه کنید:
\begin{equation}
\gamma_{i1} = \frac{w_1 \mathcal{N}(x_i; \mu_1, \sigma_1^2)}{w_1 \mathcal{N}(x_i; \mu_1, \sigma_1^2) + w_2 \mathcal{N}(x_i; \mu_2, \sigma_2^2)}
\end{equation}
و $\gamma_{i2} = 1 - \gamma_{i1}$.

\item \textbf{مرحلهٔ M (Maximization):} پارامترها را به‌روزرسانی کنید:
\begin{equation}
w_1 = \frac{\sum_{i=1}^{n} \gamma_{i1}}{n}, \quad w_2 = 1 - w_1
\end{equation}
\begin{equation}
\mu_1 = \frac{\sum_{i=1}^{n} \gamma_{i1} x_i}{\sum_{i=1}^{n} \gamma_{i1}}, \quad \mu_2 = \frac{\sum_{i=1}^{n} \gamma_{i2} x_i}{\sum_{i=1}^{n} \gamma_{i2}}
\end{equation}
\begin{equation}
\sigma_1^2 = \frac{\sum_{i=1}^{n} \gamma_{i1} (x_i - \mu_1)^2}{\sum_{i=1}^{n} \gamma_{i1}}, \quad \sigma_2^2 = \frac{\sum_{i=1}^{n} \gamma_{i2} (x_i - \mu_2)^2}{\sum_{i=1}^{n} \gamma_{i2}}
\end{equation}
\end{enumerate}

\paragraph{مرحلهٔ ۳: محاسبهٔ معیارهای اطلاعاتی}

\begin{equation}
\ell(\theta) = \sum_{i=1}^{n} \ln\left(w_1 \mathcal{N}(x_i; \mu_1, \sigma_1^2) + w_2 \mathcal{N}(x_i; \mu_2, \sigma_2^2)\right)
\end{equation}

\begin{equation}
\text{AICc} = -2\ell(\theta) + 2k + \frac{2k(k+1)}{n-k-1}, \quad k = 5
\end{equation}

\begin{equation}
\text{BIC} = -2\ell(\theta) + k\ln(n)
\end{equation}

\subsubsection{شاخص‌های تشخیص دوکوهانه‌ای}

\begin{enumerate}
\item \textbf{شاخص جدایش اشمن (Ashman's D):}
\begin{equation}
D = \frac{|\mu_2 - \mu_1|}{\sqrt{(\sigma_1^2 + \sigma_2^2)/2}}
\end{equation}
تفسیر: $D > 2$: دوکوهانه‌ای واضح، $1.5 < D < 2$: دوکوهانه‌ای متوسط، $D < 1.5$: دوکوهانه‌ای ضعیف

\item \textbf{ضریب همپوشانی (Overlap Coefficient):}
\begin{equation}
\text{OVL} = \int_{-\infty}^{\infty} \min\{f_1(x), f_2(x)\} \, dx
\end{equation}
تفسیر: OVL نزدیک به ۰: جدایش کامل، OVL نزدیک به ۱: همپوشانی کامل

\item \textbf{قدرت دوکوهانه‌ای (Bimodality Strength):}
\begin{equation}
\beta = \frac{w_1 w_2 (\mu_2 - \mu_1)^2}{\sigma_1 \sigma_2}
\end{equation}
\end{enumerate}

\subsubsection{کد کامل پیاده‌سازی در پایتون}

\begin{verbatim}
import numpy as np
from scipy.stats import norm
from scipy.special import logsumexp

class BimodalNormal:
    def __init__(self, w1=0.5, mu1=0.0, mu2=1.0, sigma1=1.0, sigma2=1.0):
        self.w1 = np.clip(w1, 0.0, 1.0)
        self.w2 = 1.0 - self.w1
        self.mu1, self.mu2 = min(mu1, mu2), max(mu1, mu2)
        self.sigma1 = max(sigma1, 1e-6)
        self.sigma2 = max(sigma2, 1e-6)
        self._dist1 = norm(self.mu1, self.sigma1)
        self._dist2 = norm(self.mu2, self.sigma2)
    
    def pdf(self, x):
        """تابع چگالی احتمال"""
        x = np.asarray(x)
        return self.w1 * self._dist1.pdf(x) + self.w2 * self._dist2.pdf(x)
    
    def logpdf(self, x):
        """لگاریتم تابع چگالی احتمال"""
        x = np.asarray(x)
        log_w = np.log([self.w1, self.w2])
        log_pdf = np.column_stack([self._dist1.logpdf(x), self._dist2.logpdf(x)])
        return logsumexp(log_pdf + log_w, axis=1)
    
    @classmethod
    def fit(cls, data, max_iter=1000, tol=1e-6):
        """برازش توزیع دوکوهانه با الگوریتم EM"""
        data = np.asarray(data).flatten()
        data = data[~np.isnan(data)]
        n = len(data)
        
        if n < 10:
            raise ValueError("حداقل ۱۰ داده برای برازش نیاز است")
        
        # مقداردهی اولیه
        mu = np.mean(data)
        sigma = np.std(data, ddof=0)
        skew = np.mean(((data - mu)/sigma)**3)
        kurt = np.mean(((data - mu)/sigma)**4) - 3
        
        if abs(skew) < 0.1:
            w1 = 0.5
            delta = sigma * np.sqrt(max(0, kurt/2)) if kurt > 0 else 0.5*sigma
            mu1, mu2 = mu - delta, mu + delta
            sigma1 = sigma2 = np.sqrt(max(0.1, sigma**2 - delta**2))
        else:
            w1 = 0.5 - 0.4 * np.clip(skew, -1, 1)
            w1 = np.clip(w1, 0.01, 0.99)
            w2 = 1 - w1
            delta = sigma * (0.5 + 0.3 * abs(skew))
            if skew > 0:
                mu1, mu2 = mu - w2*delta, mu + w1*delta
            else:
                mu1, mu2 = mu + w2*delta, mu - w1*delta
            sigma1 = sigma2 = np.sqrt(max(0.1, sigma**2 - w1*w2*delta**2))
        
        # الگوریتم EM
        for iteration in range(max_iter):
            # مرحله E
            pdf1 = norm.pdf(data, mu1, sigma1)
            pdf2 = norm.pdf(data, mu2, sigma2)
            gamma = (w1 * pdf1) / (w1 * pdf1 + w2 * pdf2 + 1e-300)
            
            # مرحله M
            w1_new = np.mean(gamma)
            w2_new = 1 - w1_new
            mu1_new = np.sum(gamma * data) / (np.sum(gamma) + 1e-300)
            mu2_new = np.sum((1-gamma) * data) / (np.sum(1-gamma) + 1e-300)
            sigma1_new = np.sqrt(np.sum(gamma * (data - mu1_new)**2) / (np.sum(gamma) + 1e-300))
            sigma2_new = np.sqrt(np.sum((1-gamma) * (data - mu2_new)**2) / (np.sum(1-gamma) + 1e-300))
            
            # بررسی همگرایی
            diff = max(abs(w1_new - w1), abs(mu1_new - mu1), abs(mu2_new - mu2),
                      abs(sigma1_new - sigma1), abs(sigma2_new - sigma2))
            
            w1, w2 = w1_new, w2_new
            mu1, mu2 = mu1_new, mu2_new
            sigma1, sigma2 = sigma1_new, sigma2_new
            
            if diff < tol:
                break
        
        # محاسبه معیارهای اطلاعاتی
        loglik = np.sum(np.log(cls(w1, mu1, mu2, sigma1, sigma2).pdf(data) + 1e-300))
        k = 5
        aic = 2*k - 2*loglik
        bic = k*np.log(n) - 2*loglik
        aicc = aic + (2*k*(k+1))/(n-k-1) if n-k-1 > 0 else aic
        
        # شاخص‌های دوکوهانه‌ای
        D = abs(mu2 - mu1) / np.sqrt((sigma1**2 + sigma2**2)/2)
        beta = w1 * w2 * (mu2 - mu1)**2 / (sigma1 * sigma2)
        
        return {
            'w1': w1, 'w2': w2,
            'mu1': mu1, 'mu2': mu2,
            'sigma1': sigma1, 'sigma2': sigma2,
            'loglik': loglik, 'aicc': aicc, 'bic': bic,
            'ashman_d': D, 'bimodality_strength': beta,
            'iterations': iteration + 1
        }
    
    @property
    def ashman_d(self):
        """شاخص جدایش اشمن"""
        return abs(self.mu2 - self.mu1) / np.sqrt((self.sigma1**2 + self.sigma2**2)/2)
    
    @property
    def bimodality_strength(self):
        """قدرت دوکوهانه‌ای"""
        return self.w1 * self.w2 * (self.mu2 - self.mu1)**2 / (self.sigma1 * self.sigma2)
    
    @property
    def overlap_coefficient(self):
        """ضریب همپوشانی (محاسبه عددی)"""
        from scipy.integrate import quad
        f1 = lambda x: norm.pdf(x, self.mu1, self.sigma1)
        f2 = lambda x: norm.pdf(x, self.mu2, self.sigma2)
        return quad(lambda x: np.minimum(f1(x), f2(x)), -np.inf, np.inf)[0]
\end{verbatim}

\subsection{توزیع پیرسون نوع III}

\subsubsection{تعریف}

توزیع پیرسون نوع III با تابع چگالی احتمال زیر تعریف می‌شود:

\begin{equation}
f(x) = \frac{1}{\Gamma(\alpha)\beta^\alpha} (x-\gamma)^{\alpha-1} \exp\left(-\frac{x-\gamma}{\beta}\right)
\end{equation}

\subsubsection{پارامترها}

\begin{itemize}
\item $\alpha > 0$: پارامتر شکل
\item $\beta > 0$: پارامتر مقیاس
\item $\gamma$: پارامتر مکان (حد پایین)
\end{itemize}

\subsubsection{روش برآورد پارامترها (روش گشتاورها)}

\begin{enumerate}
\item میانگین ($\mu$)، انحراف معیار ($\sigma$) و چولگی ($\gamma_1$) داده را محاسبه کنید.
\item $\alpha$ را محاسبه کنید:
\begin{equation}
\alpha = \frac{4}{\gamma_1^2}
\end{equation}
\item $\beta$ را محاسبه کنید:
\begin{equation}
\beta = \frac{\sigma}{\sqrt{\alpha}}
\end{equation}
\item $\gamma$ را محاسبه کنید:
\begin{equation}
\gamma = \mu - \alpha\beta
\end{equation}
\end{enumerate}

\subsubsection{کد نمونه در پایتون}

\begin{verbatim}
import numpy as np
from scipy.stats import pearson3

def fit_pearson3(data):
    mu = np.mean(data)
    sigma = np.std(data, ddof=0)
    skew = np.mean(((data - mu)/sigma)**3)
    
    # برآورد با روش گشتاورها
    skew = np.clip(skew, -2.999, 2.999)
    alpha = 4 / (skew**2 + 1e-10)
    beta = sigma / np.sqrt(alpha)
    gamma = mu - alpha * beta
    
    # در صورت چولگی منفی، داده را معکوس می‌کنیم
    if skew < 0:
        data_transformed = -data
        # ... برآورد روی داده‌های معکوس
    else:
        data_transformed = data
    
    # محاسبه درست‌نمایی
    from scipy.stats import gamma
    loglik = np.sum(gamma.logpdf(data_transformed - gamma, alpha, scale=beta))
    
    n = len(data)
    k = 3
    aic = 2*k - 2*loglik
    bic = k*np.log(n) - 2*loglik
    aicc = aic + (2*k*(k+1))/(n-k-1) if n-k-1 > 0 else aic
    
    return {'shape': alpha, 'scale': beta, 'loc': gamma,
            'loglik': loglik, 'aicc': aicc, 'bic': bic}
\end{verbatim}

\section{معیارهای انتخاب توزیع}

پس از برازش هر پنج توزیع به داده‌ها، سامانه بهترین توزیع را با استفاده از معیارهای اطلاعاتی انتخاب می‌کند.

\subsection{معیار اطلاعات آکائیکه تصحیح‌شده (AICc)}

\begin{equation}
\text{AICc} = -2\ln(\hat{L}) + 2k + \frac{2k(k+1)}{n-k-1}
\end{equation}

که در آن $\hat{L}$ بیشینه‌ی تابع درست‌نمایی، $k$ تعداد پارامترهای توزیع و $n$ حجم نمونه است.

\subsection{معیار اطلاعات بیزی (BIC)}

\begin{equation}
\text{BIC} = -2\ln(\hat{L}) + k\ln(n)
\end{equation}

\subsection{قانون انتخاب}

سامانه توزیعی را انتخاب می‌کند که کمترین مقدار AICc را داشته باشد. در صورت تساوی (تفاوت کمتر از ۲)، توزیع با تعداد پارامتر کمتر (اصل سادگی) ترجیح داده می‌شود.

\section{راهنمای افزودن توزیع جدید}

برای افزودن یک توزیع جدید به سامانه، مراحل زیر را دنبال کنید:

\begin{enumerate}
\item یک فایل پایتون جدید در پوشهٔ \texttt{plugins/distributions/} ایجاد کنید (مثلاً \texttt{my\_distribution.py}).

\item کلاسی تعریف کنید که از \texttt{DistributionPlugin} ارث‌بری کند:

\begin{verbatim}
from core.engine.distribution_plugin import DistributionPlugin

class MyDistribution(DistributionPlugin):
    name = "MyDistribution"     # نام نمایشی
    code = 5                    # کد یکتا (بعد از ۴)
    params = ["p1", "p2", "p3"] # نام پارامترها
    n_params = 3                # تعداد پارامترها
    supports_negative = True    # پشتیبانی از داده‌های منفی
    supports_zero = True        # پشتیبانی از صفر
    supports_positive = True    # پشتیبانی از داده‌های مثبت
    extreme_only = False        # فقط در حالت فرین فعال باشد؟
\end{verbatim}

\item متد \texttt{fit(self, data)} را پیاده‌سازی کنید:

\begin{verbatim}
def fit(self, data):
    # برآورد پارامترها
    p1, p2, p3 = self._estimate_parameters(data)
    
    # محاسبه درست‌نمایی
    loglik = self._compute_loglikelihood(data, p1, p2, p3)
    
    # محاسبه معیارهای اطلاعاتی
    n = len(data)
    k = self.n_params
    aic = 2*k - 2*loglik
    bic = k*np.log(n) - 2*loglik
    aicc = aic + (2*k*(k+1))/(n-k-1) if n-k-1 > 0 else aic
    
    return {
        "p1": p1, "p2": p2, "p3": p3,
        "loglik": loglik,
        "aicc": aicc,
        "bic": bic
    }
\end{verbatim}

\item سامانه به‌طور خودکار توزیع جدید را شناسایی و بارگذاری می‌کند.
\end{enumerate}
